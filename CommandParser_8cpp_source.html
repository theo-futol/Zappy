87bool CommandParser::feed(std::vector<std::unique_ptr<Client>> &clients)
88{
89 char buf[4096] = {0};
90 int bytes = recv(_client->getFd(), buf, sizeof(buf) - 1, 0);
91 if (bytes <= 0)
92 return false;
93 std::string buffer = _client->getBuffer() + std::string(buf, bytes);
94 size_t pos;
95 while ((pos = buffer.find('\n')) != std::string::npos)
96 {
97 std::string line = buffer.substr(0, pos);
98 buffer.erase(0, pos + 1);
99 if (line.empty())
100 continue;
101 Player *player = _world ? _world->getPlayerById(_client->getPlayerId()) : nullptr;
102 Logger::log("110", "Parsing : input received", {{"player_id", player ? std::to_string(player->getId()) : std::to_string(_client->getFd())}});
103 std::istringstream iss(line);
104 std::string cmd;
105 iss >> cmd;
106 auto it = _aiCommands.find(cmd);
107 int cost = (it != _aiCommands.end()) ? it->second.first : 0;
108 auto base = _commandQueue.empty() ? std::chrono::steady_clock::now() : _commandQueue.back().readyAt;
109 auto readyAt = base + std::chrono::milliseconds(cost * 1000 / _world->getTimeUnit());
110 if (cmd == "Incantation" && _client->getType() == ClientType::AI)
111 {
112 if (!player)
113 continue;
114 if (player->isFrozen())
115 {
116 Logger::log("8428", "Incantation : failed, player frozen during ritual", {{"player_id", std::to_string(player->getId())}});
117 continue;
118 }
119 if (!_commands.beginIncantation(*_client, readyAt, clients))
120 continue;
121 }
122 _commandQueue.push({line, readyAt});
123 }
124 _client->setBuffer(buffer);
125 return true;
