48def parse_server_line(line: str) -> dict[str, object]:
49 raw = _normalize_line(line)
50
51 if raw == "WELCOME":
52 return {"type": "welcome"}
53 if raw == "ok":
54 return {"type": "ok"}
55 if raw == "ko":
56 return {"type": "ko"}
57 if raw == "dead":
58 return {"type": "dead"}
59 if raw == "Elevation underway":
60 return {"type": "elevation_underway"}
61 if raw.isdigit():

63

65 if level_match is not None:
66 return {"type": "current_level", "level": int(level_match.group(1))}
67

69 if game_end_match is not None:
70 return {"type": "game_end", "team": game_end_match.group(1).strip()}

73 if broadcast_match is not None:
74 return {
75 "type": "broadcast",
76 "direction": int(broadcast_match.group(1)),
77 "message": broadcast_match.group(2),
78 }
79
80 eject_match = _EJECT_RE.match(raw)
81 if eject_match is not None:
82 return {
83 "type": "eject",
84 "direction": int(eject_match.group(1)),
85 }
86
87 if raw.startswith("[") and raw.endswith("]"):
88 if _looks_like_inventory_payload(raw):
89 return {
90 "type": "inventory",
91 "resources": parse_inventory_payload(raw),

94 "type": "look",

99 if len(tokens) == 2 and all(token.lstrip("-").isdigit() for token in tokens):
100 return {
101 "type": "map_size",
102 "width": int(tokens[0]),
103 "height": int(tokens[1]),

122def _parse_look_tile(tile: str, *, is_current_tile: bool = False) -> list[str]:
123 items: list[str] = []
124 parts = tile.strip().split()
125 index = 0
126 while index < len(parts):
127 part = parts[index]
128 if (
129 part == "player"
130 and index + 1 < len(parts)
131 and parts[index + 1].isdigit()
132 ):
133 quantity = int(parts[index + 1])
134 if is_current_tile:
135 quantity += 1
136 items.extend(["player"] * quantity)
137 index += 2
138 continue
139
140 compact_match = _COMPACT_LOOK_ITEM_RE.match(part)
141 if compact_match is None:
142 items.append(part)
143 index += 1
144 continue
145
146 resource_name = _normalize_look_item_alias(compact_match.group(1))
147 quantity = int(compact_match.group(2))
148 if resource_name not in RESOURCE_IDS and resource_name != "player":
149 items.append(part)
150 index += 1
151 continue
152 items.extend([resource_name] * quantity)
153 index += 1
154 return items

166def parse_inventory_payload(payload: str) -> dict[str, int]:
167 inner = _strip_brackets(payload)
168 resources = {name: 0 for name in RESOURCE_NAMES}
169 if inner == "":
170 return resources
171
172 for entry in inner.split(","):
173 item = entry.strip()
174 if not item:
175 continue
176 parts = item.split()
177 if len(parts) != 2:
178 raise ProtocolError(f"Invalid inventory entry: {item!r}")
179 name, value = parts
180 _validate_resource_name(name)
181 resources[name] = int(value)
182 return resources

197 inner = _strip_brackets(payload)
