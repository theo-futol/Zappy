17/**
18 * @brief Offset (in floats) of keyframe `key`'s value inside a channel's values array.
19 *
20 * Cubic-spline output stores three vectors per keyframe (in-tangent, value, out-tangent);
21 * we keep only the middle value, so cubic strides by 3 and skips the in-tangent.
22 */
23std::size_t valueOffset(const ModelAnimationChannel &channel, std::size_t key)
24{
25 std::size_t stride = (channel.interp == AnimationInterp::Cubic) ? 3 : 1;
26 std::size_t bias = (channel.interp == AnimationInterp::Cubic) ? 1 : 0;
27
28 return (key * stride + bias) * channel.components;
29}

31/** @brief Reads keyframe `key` as a vec3 (translation/scale). */
32Vec3 readVec3(const ModelAnimationChannel &channel, std::size_t key)

34 std::size_t o = valueOffset(channel, key);

36 return Vec3(channel.values[o], channel.values[o + 1], channel.values[o + 2]);

39/** @brief Reads keyframe `key` as a quaternion (rotation), converting glTF xyzw to glm wxyz. */
40glm::quat readQuat(const ModelAnimationChannel &channel, std::size_t key)
41{
42 std::size_t o = valueOffset(channel, key);
43
44 return glm::quat(channel.values[o + 3], channel.values[o], channel.values[o + 1], channel.values[o + 2]);

47/**
48 * @brief Samples a channel at `time` and writes the result into the node's TRS slots.
49 * @param channel Source channel.
50 * @param time Time within the clip (seconds).
51 * @param translations Per-node translation, overwritten if the channel animates translation.
52 * @param rotations Per-node rotation, overwritten if the channel animates rotation.
53 * @param scales Per-node scale, overwritten if the channel animates scale.
54 */
55void sampleChannel(const ModelAnimationChannel &channel, float time, std::vector<Vec3> &translations, std::vector<glm::quat> &rotations, std::vector<Vec3> &scales)
56{
57 std::size_t count = channel.times.size();
58
59 if (count == 0 || channel.node >= translations.size())
60 return;
61
62 std::size_t key = 0;
63
64 while (key + 1 < count && channel.times[key + 1] < time)
65 ++key;
66
67 std::size_t next = (key + 1 < count) ? key + 1 : key;
68 float span = channel.times[next] - channel.times[key];
69 float alpha = (span > 0.0f) ? glm::clamp((time - channel.times[key]) / span, 0.0f, 1.0f) : 0.0f;
70 bool step = (channel.interp == AnimationInterp::Step);
71
72 if (channel.path == AnimationPath::Rotation)
73 {
74 glm::quat value = step ? readQuat(channel, key) : glm::slerp(readQuat(channel, key), readQuat(channel, next), alpha);
75
76 rotations[channel.node] = glm::normalize(value);
77 return;
78 }
79
80 Vec3 value = step ? readVec3(channel, key) : glm::mix(readVec3(channel, key), readVec3(channel, next), alpha);
81
82 if (channel.path == AnimationPath::Translation)
83 translations[channel.node] = value;
84 else
85 scales[channel.node] = value;
86}

130MeshData RenderModel::interleave(const ModelPrimitive &primitive)
131{
132 MeshData data;
133 std::size_t count = primitive.positions.size() / 3;
134 bool skinned = primitive.joints.size() >= count * 4 && primitive.weights.size() >= count * 4;
135
136 data.vertices.reserve(count * 16);
137 for (std::size_t i = 0; i < count; ++i)
138 {
139 data.vertices.push_back(primitive.positions[i * 3 + 0]);
140 data.vertices.push_back(primitive.positions[i * 3 + 1]);
141 data.vertices.push_back(primitive.positions[i * 3 + 2]);
142 if (primitive.normals.size() >= (i + 1) * 3)
143 {
144 data.vertices.push_back(primitive.normals[i * 3 + 0]);
145 data.vertices.push_back(primitive.normals[i * 3 + 1]);
146 data.vertices.push_back(primitive.normals[i * 3 + 2]);
147 }
148 else
149 {
150 data.vertices.push_back(0.0f);
151 data.vertices.push_back(1.0f);
152 data.vertices.push_back(0.0f);
153 }
154 if (primitive.texcoords.size() >= (i + 1) * 2)
155 {
156 data.vertices.push_back(primitive.texcoords[i * 2 + 0]);
157 data.vertices.push_back(primitive.texcoords[i * 2 + 1]);
158 }
159 else
160 {
161 data.vertices.push_back(0.0f);
162 data.vertices.push_back(0.0f);
163 }
164 for (std::size_t k = 0; k < 4; ++k)
165 data.vertices.push_back(skinned ? primitive.joints[i * 4 + k] : 0.0f);
166 for (std::size_t k = 0; k < 4; ++k)
167 data.vertices.push_back(skinned ? primitive.weights[i * 4 + k] : 0.0f);
168 }
169 data.indices = primitive.indices;
170 return data;
171}

222{
223 const ModelNode &node = model.nodes[nodeIndex];

227 {

237 Vec3 worldPoint(world * Vec4(point, 1.0f));
238
239 boundsMin = glm::min(boundsMin, worldPoint);
240 boundsMax = glm::max(boundsMax, worldPoint);
241 }
242 }
243 for (std::size_t child : node.children)
244 flattenNode(model, child, world, boundsMin, boundsMax);

257{

302std::vector<Mat4> RenderModel::poseJoints(std::size_t animation, float time) const
303{
304 if (animation >= _animations.size())
305 return _bindJoints;
306
307 std::vector<Vec3> translations(_nodes.size());
308 std::vector<glm::quat> rotations(_nodes.size());
309 std::vector<Vec3> scales(_nodes.size());
310 std::vector<bool> animated(_nodes.size(), false);
311
312 for (std::size_t n = 0; n < _nodes.size(); ++n)
313 {
314 translations[n] = _nodes[n].translation;
315 rotations[n] = _nodes[n].rotation;
316 scales[n] = _nodes[n].scale;
317 }
318 for (const ModelAnimationChannel &channel : _animations[animation].channels)
319 {
320 sampleChannel(channel, time, translations, rotations, scales);
321 if (channel.node < animated.size())
322 animated[channel.node] = true;
323 }
324
325 std::vector<Mat4> locals(_nodes.size(), Mat4(1.0f));
326
327 for (std::size_t n = 0; n < _nodes.size(); ++n)
328 locals[n] = animated[n] ? glm::translate(Mat4(1.0f), translations[n]) * glm::mat4_cast(rotations[n]) * glm::scale(Mat4(1.0f), scales[n]) : _nodes[n].transform;
329
330 std::vector<Mat4> globals(_nodes.size(), Mat4(1.0f));
331
332 for (std::size_t root : _roots)
333 poseGlobals(root, Mat4(1.0f), locals, globals);
334
335 std::vector<Mat4> joints(_skin.joints.size());
336
337 for (std::size_t k = 0; k < _skin.joints.size(); ++k)
338 joints[k] = globals[_skin.joints[k]] * _skin.inverseBind[k];
339 return joints;
340}

342void RenderModel::accumulateSkinnedBounds(const Model &model, Vec3 &boundsMin, Vec3 &boundsMax) const
343{
344 for (std::size_t p = 0; p < model.primitives.size(); ++p)
345 {
346 const ModelPrimitive &primitive = model.primitives[p];
347 std::size_t count = primitive.positions.size() / 3;
348 bool hasSkin = primitive.joints.size() >= count * 4 && primitive.weights.size() >= count * 4;
349
350 if (!hasSkin)
351 {
352 boundsMin = glm::min(boundsMin, _localMin[p]);
353 boundsMax = glm::max(boundsMax, _localMax[p]);
354 continue;
355 }
356 for (std::size_t v = 0; v < count; ++v)
357 {
358 Vec4 position(primitive.positions[v * 3], primitive.positions[v * 3 + 1], primitive.positions[v * 3 + 2], 1.0f);
359 Mat4 skin(0.0f);
360
361 for (std::size_t k = 0; k < 4; ++k)
362 {
363 float weight = primitive.weights[v * 4 + k];
364 std::size_t joint = static_cast<std::size_t>(primitive.joints[v * 4 + k]);
365
366 if (weight > 0.0f && joint < _bindJoints.size())
367 skin += weight * _bindJoints[joint];
368 }
369 Vec3 posed(skin * position);
370
371 boundsMin = glm::min(boundsMin, posed);
372 boundsMax = glm::max(boundsMax, posed);
373 }
374 }

388{
