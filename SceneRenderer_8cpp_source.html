79{
80 Vec3 up = glm::normalize(point.normal);

116void SceneRenderer::drawBroadcasts(Shader &shader, const RenderContext &context)
117{
118 for (const GameState::Broadcast &broadcast : context.state.broadcasts())
119 if (broadcast.sequence > _seenBroadcastSeq)
120 {
121 _seenBroadcastSeq = broadcast.sequence;
122 _pings.push_back(BroadcastPing{broadcast.origin, context.time});
123 }
124 for (const BroadcastPing &ping : _pings)
125 {
126 float progress = (context.time - ping.start) / BroadcastDuration;
127
128 if (progress < 0.0f || progress > 1.0f)
129 continue;
130
131 WorldPoint point = context.mapping.toWorld(ping.origin.x, ping.origin.y);
132 float diameter = glm::mix(0.4f, BroadcastMaxTiles, progress);
133
134 drawRing(shader, point, diameter, BroadcastColor);
135 }
136 _pings.erase(std::remove_if(_pings.begin(), _pings.end(), [&](const BroadcastPing &ping) { return context.time - ping.start > BroadcastDuration; }), _pings.end());

250SceneRenderer::EntityAnim &SceneRenderer::updateMovement(const EntityKey &key, GridPosition position, float time)
251{
252 EntityAnim &state = _entityAnim[key];
253
254 if (!state.initialized)
255 {
256 state.initialized = true;
257 state.fromPos = position;
258 state.toPos = position;
259 }
260 if (position.x != state.toPos.x || position.y != state.toPos.y)
261 {
262 int dx = position.x - state.toPos.x;
263 int dy = position.y - state.toPos.y;
264 bool adjacent = (dx >= -1 && dx <= 1 && dy >= -1 && dy <= 1);
265
266 state.fromPos = adjacent ? state.toPos : position;
267 state.toPos = position;
268 state.moveStart = time;
269 state.sliding = adjacent;
270 }
271 return state;
272}

274WorldPoint SceneRenderer::entityWorld(const EntityAnim &state, const IProjection &mapping, float time) const
275{
276 WorldPoint to = mapping.toWorld(state.toPos.x, state.toPos.y);
277 float alpha = (state.sliding && MoveDuration > 0.0f) ? glm::clamp((time - state.moveStart) / MoveDuration, 0.0f, 1.0f) : 1.0f;
278
279 if (alpha >= 1.0f)
280 return to;
281
282 WorldPoint from = mapping.toWorld(state.fromPos.x, state.fromPos.y);
283 WorldPoint result;
284
285 result.position = glm::mix(from.position, to.position, alpha);
286 result.normal = glm::normalize(glm::mix(from.normal, to.normal, alpha));
287 result.tangent = glm::normalize(glm::mix(from.tangent, to.tangent, alpha));
288 return result;
289}

291bool SceneRenderer::transformerPose(EntityAnim &state, const RenderModel &model, float time, int level, int &clip, float &poseTime)
292{
293 int vehicleClip = model.animationIndex(VehicleClipName);
294 int robotClip = model.animationIndex(RobotClipName);
295 int heroClip = model.animationIndex(HeroClipName);
296
297 if (vehicleClip < 0 || robotClip < 0)
298 return false;
299 if (heroClip < 0)
300 heroClip = robotClip;
301 if (state.lastLevel < 0)
302 state.lastLevel = level;
303 else if (level > state.lastLevel)
304 {
305 state.clip = heroClip;
306 state.clipStart = time;
307 state.vehicle = false;
308 }
309 state.lastLevel = level;
310
311 bool wantVehicle = (time - state.moveStart) < IdleBeforeRobot;
312
313 if (state.clip < 0 && wantVehicle != state.vehicle)
314 {
315 state.vehicle = wantVehicle;
316 state.clip = wantVehicle ? vehicleClip : robotClip;
317 state.clipStart = time;
318 }
319 if (state.clip >= 0 && time - state.clipStart >= model.animationDuration(static_cast<std::size_t>(state.clip)))
320 state.clip = -1;
321
322 int settledClip = state.vehicle ? vehicleClip : robotClip;
323
324 if (state.clip >= 0)
325 {
326 clip = state.clip;
327 poseTime = time - state.clipStart;
328 }
329 else
330 {
331 clip = settledClip;
332 poseTime = model.animationDuration(static_cast<std::size_t>(settledClip));
333 }
334 return true;
335}

337void SceneRenderer::render(const RenderContext &context)
338{
339 Shader *shader = _assets.shader("phong");
340
341 if (shader == nullptr)
342 return;
343 if (!context.state.winner().empty())
344 {
345 drawVictory(context);
346 return;
347 }
348
349 const Map &map = context.state.map();
350 Vec3 eye = Vec3(glm::inverse(context.view)[3]);
351 float outer = _torusMajor + _torusMinor;
352 float span = static_cast<float>(std::max(map.width(), map.height()));
353 Vec3 mapCenter(static_cast<float>(map.width()) * 0.5f, 0.0f, static_cast<float>(map.height()) * 0.5f);
354 Vec3 lightPos = _torusWorld ? Vec3(outer, 2.0f * outer, outer) : mapCenter + Vec3(span * 0.5f, span + 1.0f, span * 0.5f);
355
356 shader->use();
357 shader->setUniform("uView", context.view);
358 shader->setUniform("uProjection", context.projection);
359 shader->setUniform("uLightPos", lightPos);
360 shader->setUniform("uViewPos", eye);
361 shader->setUniform("uLightColor", LightColor);
362 shader->setUniform("uTexture", 0);
363 shader->setUniform("uBaseColor", ModelColor);
364
365 Mat4 groundBase;
366
367 if (_torusWorld)
368 {
369 Mat4 lay = glm::rotate(Mat4(1.0f), glm::radians(90.0f), Vec3(1.0f, 0.0f, 0.0f));
370
371 groundBase = lay * glm::scale(Mat4(1.0f), Vec3(2.0f * outer)) * glm::translate(Mat4(1.0f), -_ground->center()) * _ground->unitTransform();
372 }
373 else
374 {
375 Vec3 groundCenter(static_cast<float>(map.width() - 1) * 0.5f, 0.0f, static_cast<float>(map.height() - 1) * 0.5f);
376 Vec3 groundScale(static_cast<float>(map.width()), 1.0f, static_cast<float>(map.height()));
377
378 groundBase = glm::translate(Mat4(1.0f), groundCenter) * glm::scale(Mat4(1.0f), groundScale) * _ground->footprintTransform();
379 }
380 std::vector<Mat4> groundBases{groundBase};
381 std::size_t themeCount = _golems.size();
382
383 _ground->drawInstanced(*shader, groundBases);
384 for (const std::pair<const EntityKey, std::unique_ptr<IEntity>> &entry : context.state.entities())
385 {
386 const IEntity &entity = *entry.second;
387 EntityAnim &state = updateMovement(entry.first, entity.position(), context.time);
388 WorldPoint point = entityWorld(state, context.mapping, context.time);
389 std::size_t theme = context.state.teamIndex(entity.team()) % themeCount;
390 bool isEgg = entity.getEntityType() == "egg";
391 bool incanting = map.at(entity.position().x, entity.position().y).incanting();
392 const RenderModel &model = isEgg ? *_eggs[theme] : *_golems[theme];
393 float yaw = orientationYaw(entity.orientation()) * (_torusWorld ? 1.0f : -1.0f);
394 float spin = (incanting && !isEgg) ? std::fmod(context.time * IncantSpinSpeed, 360.0f) : 0.0f;
395 Mat4 facing = glm::rotate(Mat4(1.0f), glm::radians(yaw + spin), Vec3(0.0f, 1.0f, 0.0f));
396 Mat4 base = glm::translate(Mat4(1.0f), point.position) * tileFrame(point) * facing * model.unitTransform();
397
398 if (model.skinned())
399 {
400 int clip = 0;
401 float poseTime = 0.0f;
402
403 if (isEgg || !transformerPose(state, model, context.time, entity.level(), clip, poseTime))
404 {
405 float duration = model.animationDuration(0);
406
407 clip = 0;
408 poseTime = (duration > 0.0f) ? std::fmod(context.time, duration) : 0.0f;
409 }
410 model.drawSkinned(*shader, base, model.poseJoints(static_cast<std::size_t>(clip), poseTime));
411 }
412 else
413 model.drawInstanced(*shader, std::vector<Mat4>{base});
414 }
415
416 std::vector<std::vector<Mat4>> resourceBases(ResourceSet::Count);
417
418 for (int gridY = 0; gridY < map.height(); ++gridY)
419 for (int gridX = 0; gridX < map.width(); ++gridX)
420 {
421 const Tile &tile = map.at(gridX, gridY);
422 WorldPoint point = context.mapping.toWorld(gridX, gridY);
423 Mat4 frame = tileFrame(point);
424
425 if (tile.incanting())
426 {
427 float pulse = 0.45f + 0.55f * std::sin(context.time * 6.0f);
428
429 drawHighlight(*shader, point, IncantColor * pulse);
430 }
431 for (std::size_t type = 0; type < ResourceSet::Count; ++type)
432 {
433 if (_resources[type] == nullptr || tile.resources().get(static_cast<ResourceType>(type)) <= 0)
434 continue;
435 Mat4 tilt = glm::rotate(Mat4(1.0f), glm::radians(resourceTilt(type)), Vec3(1.0f, 0.0f, 0.0f));
436 Mat4 base = glm::translate(Mat4(1.0f), point.position) * frame * glm::translate(Mat4(1.0f), resourceSpot(type)) * glm::scale(Mat4(1.0f), Vec3(ResourceScale)) *
437 tilt * _resources[type]->unitTransform();
438
439 resourceBases[type].push_back(base);
440 }
441 }
442 for (std::size_t type = 0; type < ResourceSet::Count; ++type)
443 if (_resources[type] != nullptr)
444 {
445 shader->setUniform("uBaseColor", resourceTint(type));
446 _resources[type]->drawInstanced(*shader, resourceBases[type]);
447 }
448
449 const IEntity *selected = context.state.selectedEntity();
450
451 if (context.state.hasHoveredTile())
452 {
453 GridPosition hovered = context.state.hoveredTile();
454
455 drawHighlight(*shader, context.mapping.toWorld(hovered.x, hovered.y), HoverColor);
456 }
457 if (selected != nullptr)
458 drawHighlight(*shader, context.mapping.toWorld(selected->position().x, selected->position().y), SelectColor);
459 drawBroadcasts(*shader, context);
460}
