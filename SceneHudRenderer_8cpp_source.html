198 }
199 if (isPlayer)

295std::string SceneHudRenderer::handleClick(double px, double py, const GameState &state) const
296{
297 float fx = static_cast<float>(px);
298 float fy = static_cast<float>(_height) - static_cast<float>(py);
299 int base = state.timeUnit() < MinTimeUnit ? MinTimeUnit : state.timeUnit();
300
301 if (contains(minusButton(), fx, fy))
302 return CommandBuilder::setTimeUnit(std::max(MinTimeUnit, base / 2));
303 if (contains(plusButton(), fx, fy))
304 return CommandBuilder::setTimeUnit(std::min(MaxTimeUnit, base * 2));
305 return "";
306}
