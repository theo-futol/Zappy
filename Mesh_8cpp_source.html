34void Mesh::upload(const std::vector<float> &vertices, const std::vector<unsigned int> &indices, const std::vector<unsigned int> &attributeSizes)
35{
36 if (indices.empty() || vertices.empty())
37 throw MeshException("Mesh error: empty vertices or indices");
38 if (attributeSizes.empty())
39 throw MeshException("Mesh error: empty attribute layout");
40
41 unsigned int stride = 0;
42
43 for (unsigned int size : attributeSizes)
44 stride += size;
45 if (stride == 0 || vertices.size() % stride != 0)
46 throw MeshException("Mesh error: vertex count is not a multiple of the attribute stride");
47
48 _indexCount = static_cast<GLsizei>(indices.size());
49 _attributeCount = static_cast<GLuint>(attributeSizes.size());
50 glBindVertexArray(_vao);
51 glBindBuffer(GL_ARRAY_BUFFER, _vbo);
52 glBufferData(GL_ARRAY_BUFFER, vertices.size() * sizeof(float), vertices.data(), GL_STATIC_DRAW);
53 glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, _ebo);
54 glBufferData(GL_ELEMENT_ARRAY_BUFFER, indices.size() * sizeof(unsigned int), indices.data(), GL_STATIC_DRAW);
55
56 GLsizei strideBytes = static_cast<GLsizei>(stride * sizeof(float));
57 std::size_t offset = 0;
58
59 for (std::size_t location = 0; location < attributeSizes.size(); ++location)
60 {
61 glVertexAttribPointer(static_cast<GLuint>(location), static_cast<GLint>(attributeSizes[location]), GL_FLOAT, GL_FALSE, strideBytes, reinterpret_cast<const void *>(offset));
62 glEnableVertexAttribArray(static_cast<GLuint>(location));
63 offset += attributeSizes[location] * sizeof(float);
64 }
65 glBindVertexArray(0);
66}
