32void MapRenderer::render(const RenderContext &context)
33{
34 Shader *shader = _assets.shader("basic");
35 Mesh *mesh = _assets.mesh("quad");
36 const Map &map = context.state.map();
37
38 if (shader == nullptr || mesh == nullptr)
39 return;
40 shader->use();
41 shader->setUniform("uView", context.view);
42 shader->setUniform("uProjection", context.projection);
43
44 for (int gridY = 0; gridY < map.height(); ++gridY)
45 {
46 for (int gridX = 0; gridX < map.width(); ++gridX)
47 {
48 WorldPoint point = context.mapping.toWorld(gridX, gridY);
49 const Tile &tile = map.at(gridX, gridY);
50 Vec3 color = tileColor(gridX, gridY);
51
52 if (tile.incanting())
53 {
54 float pulse = 0.45f + 0.55f * std::sin(context.time * 6.0f);
55
56 color = glm::mix(color, IncantColor, pulse);
57 }
58 shader->setUniform("uModel", glm::translate(Mat4(1.0f), point.position));
59 shader->setUniform("uColor", color);
60 mesh->draw();
61 drawResources(*shader, *mesh, tile, point.position);
62 }
63 }
64 drawBroadcasts(*shader, *mesh, context);
65}

67void MapRenderer::drawBroadcasts(Shader &shader, Mesh &mesh, const RenderContext &context)
68{
69 (void)mesh;
70 for (const GameState::Broadcast &broadcast : context.state.broadcasts())
71 if (broadcast.sequence > _seenBroadcastSeq)
72 {
73 _seenBroadcastSeq = broadcast.sequence;
74 _pings.push_back(BroadcastPing{broadcast.origin, context.time});
75 }
76 if (_ring == nullptr)
77 return;
78 for (const BroadcastPing &ping : _pings)
79 {
80 float progress = (context.time - ping.start) / BroadcastDuration;
81
82 if (progress < 0.0f || progress > 1.0f)
83 continue;
84
85 WorldPoint point = context.mapping.toWorld(ping.origin.x, ping.origin.y);
86 float diameter = glm::mix(0.4f, BroadcastMaxTiles, progress);
87 Mat4 model = glm::scale(glm::translate(Mat4(1.0f), point.position), Vec3(diameter, diameter, 1.0f));
88
89 shader.setUniform("uModel", model);
90 shader.setUniform("uColor", BroadcastColor);
91 _ring->draw();
92 }
93 _pings.erase(std::remove_if(_pings.begin(), _pings.end(), [&](const BroadcastPing &ping) { return context.time - ping.start > BroadcastDuration; }), _pings.end());
94}
