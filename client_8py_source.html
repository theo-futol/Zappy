1"""Synchronous TCP client for a single Zappy AI player.
2
3Every command is sent and immediately waited on: this trades the protocol's
4theoretical 10-command pipelining for a client that is trivial to follow and
5never accumulates stale local state. Round-trips are cheap relative to the
6in-game action costs (default f=100), so this is not a real performance
7concern in practice.
8"""

65 def connect(self) -> None:
66 self.socket = socket.create_connection((self.host, self.port))
67 self._log(f"[connect] {self.host}:{self.port}")
68
69 event = self._read_event()
70 if event["type"] == "game_end":
71 self.game_over = True
72 self.winner_team = str(event["team"])
73 self._log(f"[state] fin de partie: {self.winner_team}")
74 return
75 if event["type"] != "welcome":
76 raise RuntimeError(f"Handshake invalide: {event!r}")
77
78 self._send_raw(self.team_name)
79 self._log(f"[send] {self.team_name}")
80
81 slots_event = self._read_event()
82 if slots_event["type"] == "ko":
83 raise RuntimeError(f"Equipe refusee par le serveur: {self.team_name}")
84 if slots_event["type"] != "available_slots":
85 raise RuntimeError(f"Slots attendus, recu: {slots_event!r}")
86 self.available_slots = int(slots_event["value"])
87
88 size_event = self._read_event()
89 if size_event["type"] != "map_size":
90 raise RuntimeError(f"Taille de carte attendue, recu: {size_event!r}")
91 self.map_width = int(size_event["width"])
92 self.map_height = int(size_event["height"])

106 def run(self) -> None:
107 self.connect()
108 if self.game_over:
109 return
110
111 strategy = Strategy(self, self.team_name)
112 try:
113 while not self.dead and not self.game_over:
114 if not strategy.step():
115 break
116 finally:
117 self.close()
118
119 if self.dead:
120 print(f"dead level={strategy.level} food={self.last_food}", file=sys.stderr)

184 def _send_and_wait(self, command: str, terminal_types: set[str]) -> dict[str, Any]:
185 if self.dead or self.game_over:
186 return {"type": "dead" if self.dead else "game_end"}
187 self._send_raw(command)
188 self._log(f"[send] {command}")
189 return self._wait_for(terminal_types)

191 def _wait_for(self, terminal_types: set[str]) -> dict[str, Any]:
192 if self.dead or self.game_over:
193 return {"type": "dead" if self.dead else "game_end"}
194
195 while True:
196 event = self._read_event()
197 event_type = str(event["type"])
198
199 if event_type == "broadcast":
200 self._pending_messages.append((int(event["direction"]), str(event["message"])))
201 continue
202 if event_type == "eject":
203 continue
204 if event_type == "dead":
205 self.dead = True
206 return event
207 if event_type == "game_end":
208 self.game_over = True
209 self.winner_team = str(event["team"])
210 return event
211 if event_type == "elevation_underway":
212 if "elevation_underway" in terminal_types:
213 return event
214 # Frozen by someone else's ritual: our own queued command (if
215 # any) gets no response at all until it resolves.
216 self._frozen_for_ritual = True
217 continue
218 if self._frozen_for_ritual and event_type in ("ko", "current_level"):
219 # This is the ritual's resolution, not our command's answer
220 # - the real one (if any) still follows once unfrozen.
221 if event_type == "current_level":
222 self.pending_level = int(event["level"])
223 self._frozen_for_ritual = False
224 continue
225 if event_type == "current_level":
226 # Pushed to every participant of a successful incantation,
227 # not just whoever sent the Incantation command.
228 self.pending_level = int(event["level"])
229 if "current_level" in terminal_types:
230 return event
231 continue
232 if event_type in terminal_types:
233 return event
234
235 self._log(f"[warn] evenement inattendu ignore: {event}")

237 def _send_raw(self, text: str) -> None:
238 if self.socket is None:
239 raise RuntimeError("Socket non connectee.")

242 def _read_event(self) -> dict[str, Any]:
243 while True:
244 raw_line = self._read_line()
245 if raw_line == "":
246 continue
247
248 self._log(f"[recv] {raw_line}")
249 try:
250 return parse_server_line(raw_line)
251 except ProtocolError as exc:
252 self._log(f"[warn] ligne non supportee: {exc}")
253 return {"type": "raw", "line": raw_line}

255 def _read_line(self) -> str:
256 if self.socket is None:
257 raise RuntimeError("Socket non connectee.")
258
259 while True:
260 if self.ready_lines:
261 return self.ready_lines.pop(0)
262
263 chunk = self.socket.recv(4096)
264 if not chunk:
265 raise ConnectionError("Connexion fermee par le serveur.")
266
267 self.ready_lines.extend(self.buffer.feed(chunk))
