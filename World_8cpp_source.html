19/// @brief Generates resources on the map at regular intervals.
20/// Via the formula : map_width * map_height * density
21/// @note The density and generation logic can be adjusted based on game requirements.
22/// This function should be called periodically, e.g., every 20 seconds, to simulate resource regeneration.
23/// The density of each ressource are the following:
24/// food: 0.5, linemate: 0.3, deraumere: 0.5, sibur: 0.1, mendiane: 0.1, phiras: 0.08, thystame: 0.05
25void World::ressourcePassiveGeneration()
26{
27 const std::vector<std::pair<ItemType, double>> resourceDensity = {{ItemType::FOOD, 0.5}, {ItemType::LINEMATE, 0.3}, {ItemType::DERAUMERE, 0.5}, {ItemType::SIBUR, 0.1},
28 {ItemType::MENDIANE, 0.1}, {ItemType::PHIRAS, 0.08}, {ItemType::THYSTAME, 0.05}};
29 int totalTiles = _map.size() * (_map.empty() ? 0 : _map[0].size());
30
31 for (const auto &[type, density] : resourceDensity)
32 {
33 int totalToGenerate = static_cast<int>(totalTiles * density);
34 for (int i = 0; i < totalToGenerate; ++i)
35 {
36 int x = rand() % _map.size();
37 int y = rand() % (_map.empty() ? 1 : _map[0].size());
38 std::vector<std::pair<zappy::ItemType, int>> &tileCoords = _map[x][y]._items;
39 auto it = std::find_if(tileCoords.begin(), tileCoords.end(), [type](const std::pair<ItemType, int> &item) { return item.first == type; });
40 if (it != tileCoords.end())
41 it->second += 1;
42 }
43 }
44}

178{
179 std::vector<int> deadFds;
180
181 for (const auto &player : _players)
182 {
183 if (player->getInventory().getItemCount(ItemType::FOOD) <= 0)
184 {
185 if (_broadcastQueue)

187 player->setState(PlayerState::DEAD);
188 deadFds.push_back(player->getFd());

227bool World::isIncantationValid(int x, int y, int level)
228{
229 const ElevationRequirement *requirement = getElevationRequirement(level);
230 tile *tilePtr = getTileAt({x, y});
231
232 if (!requirement || !tilePtr)
233 return false;
234 if (static_cast<int>(getPlayersOnTileAtLevel(x, y, level).size()) < requirement->players)
235 return false;
236 for (const auto &[stone, needed] : requirement->stones)
237 {
238 int available = 0;
239 for (const auto &item : tilePtr->_items)
240 if (item.first == stone)
241 {
242 available = item.second;
243 break;
244 }
245 if (available < needed)
246 return false;
247 }
248 return true;
249}

267bool World::checkWinningCondition()
268{
269 for (auto &team : _teams)
270 {
271 int playerCounter = 0;
272 if (team->_slotsOccupied >= 6)
273 {
274 for (auto &player : _players)
275 {
276 if (player->getTeam()._name != team->_name)
277 continue;
278 if (player->getLevel() >= 8)
279 playerCounter++;
280 }
281 if (playerCounter >= 6)
282 {
283 _broadcastQueue->push("seg " + team->_name + "\n");
284 team->_hasWin = true;
285 return true;
286 }
287 }
288 }
289 return false;

292void World::removePlayer(int fd)
293{
294 auto it = std::find_if(_players.begin(), _players.end(), [fd](const std::unique_ptr<Player> &player) { return player->getFd() == fd; });
295
296 if (it == _players.end())
297 return;
298 removePlayerFromTile(it->get(), (*it)->getPosition());
299 // setState(DEAD) already frees the team slot (see Player::setState); avoid freeing it twice.
300 if ((*it)->getState() != PlayerState::DEAD)
301 (*it)->getTeam().removePlayer();
302 _players.erase(it);
303}
