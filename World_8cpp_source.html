63/// @brief Spec-accurate algorithm: tops every resource up to map_width * map_height *
64/// density (at least 1, so Trantor always has one of each on the floor), never
65/// generating past the target nor when already there. Placement is purely random
66/// (independent random tile per unit) - a given tile may end up with nothing.
67/// food: 0.5, linemate: 0.3, deraumere: 0.15, sibur: 0.1, mendiane: 0.1,
68/// phiras: 0.08, thystame: 0.05
69void World::resourcePassiveGenerationEven()
70{
71 const std::vector<std::pair<ItemType, double>> resourceDensity = {{ItemType::FOOD, 0.5}, {ItemType::LINEMATE, 0.3}, {ItemType::DERAUMERE, 0.15}, {ItemType::SIBUR, 0.1},
72 {ItemType::MENDIANE, 0.1}, {ItemType::PHIRAS, 0.08}, {ItemType::THYSTAME, 0.05}};
73 if (_map.empty() || _map[0].empty())
74 return;
75 int width = static_cast<int>(_map.size());
76 int height = static_cast<int>(_map[0].size());
77 std::set<std::pair<int, int>> changedTiles;
78
79 for (const auto &[type, density] : resourceDensity)
80 {
81 int target = std::max(1, static_cast<int>(width * height * density));
82 int current = 0;
83 for (const auto &column : _map)
84 for (const auto &tileAt : column)
85 {
86 auto it = std::find_if(tileAt._items.begin(), tileAt._items.end(), [type](const std::pair<ItemType, int> &item) { return item.first == type; });
87 if (it != tileAt._items.end())
88 current += it->second;
89 }
90 int toAdd = target - current;
91 if (toAdd <= 0)
92 continue;
93
94 for (int i = 0; i < toAdd; ++i)
95 {
96 int x = rand() % width;
97 int y = rand() % height;
98 std::vector<std::pair<ItemType, int>> &tileItems = _map[x][y]._items;
99 auto it = std::find_if(tileItems.begin(), tileItems.end(), [type](const std::pair<ItemType, int> &item) { return item.first == type; });
100 if (it != tileItems.end())
101 {
102 it->second += 1;
103 changedTiles.emplace(x, y);
104 }
105 }
106 }
107 for (const auto &[x, y] : changedTiles)
108 broadcastTileContent({x, y});
109}

235{
236 std::shared_ptr<Team> team = getTeamByName(teamName);
237 if (!team)
238 return -1;
239 std::vector<std::pair<position, int>> hatchable;

350bool World::isIncantationValid(int x, int y, int level)
351{
352 const ElevationRequirement *requirement = getElevationRequirement(level);
353 tile *tilePtr = getTileAt({x, y});
354
355 if (!requirement || !tilePtr)
356 return false;
357 int playersAtLevel = static_cast<int>(getPlayersOnTileAtLevel(x, y, level).size());
358 if (playersAtLevel < requirement->players)
359 {
360 Logger::log("8422", "Incantation : failed, conditions not met", {{"x", std::to_string(x)}, {"y", std::to_string(y)}});
361 return false;
362 }
363 for (const auto &[stone, needed] : requirement->stones)
364 {
365 int available = 0;
366 for (const auto &item : tilePtr->_items)
367 if (item.first == stone)
368 {
369 available = item.second;
370 break;
371 }
372 if (available < needed)
373 {
374 Logger::log("8422", "Incantation : failed, conditions not met", {{"x", std::to_string(x)}, {"y", std::to_string(y)}});
375 return false;
376 }
377 }
378 return true;
379}
