89 async def run(self) -> None:
90 self._started_at = time.monotonic()
91 self._log(
92 "launcher started "
93 f"host={self.host} port={self.port} "
94 f"teams={','.join(self.teams)}"
95 )
96
97 last_status_at = 0.0
98 loop = asyncio.get_running_loop()
99
100 try:
101 while not self._stopping:
102 now = loop.time()
103 await self._maybe_spawn_clients(now)
104
105 if now - last_status_at >= self.status_interval:
106 self._log(self._build_status_line())
107 last_status_at = now
108
109 await asyncio.sleep(self.poll_interval)
110 finally:
111 await self._stop_all_clients()

113 async def _maybe_spawn_clients(self, now: float) -> None:
114 for runtime in self._team_runtimes.values():
115 if len(runtime.active_clients) >= runtime.target_count:
116 continue
117 if now < runtime.next_spawn_at:
118 continue
119
120 await self._spawn_one_client(runtime, now)
121 runtime.next_spawn_at = now + self.spawn_interval

166 async def _watch_client(
167 self,
168 runtime: TeamRuntime,
169 managed_client: ManagedClient,
170 ) -> None:
171 stderr_text = ""
172 if managed_client.process.stderr is not None:
173 stderr_output = await managed_client.process.stderr.read()
174 stderr_text = stderr_output.decode("utf-8", errors="replace").strip()
175
176 return_code = await managed_client.process.wait()
177 runtime.active_clients.pop(managed_client.instance_id, None)
178 now = asyncio.get_running_loop().time()
179 age = max(0.0, now - managed_client.spawned_at)
180
181 if self._stopping:
182 return
183
184 if return_code == GAME_OVER_EXIT_CODE:
185 self._log(
186 f"game over received from {runtime.team}#{managed_client.instance_id}, "
187 "stopping launcher"
188 )
189 self._stopping = True
190 return
191
192 reason = self._summarize_stderr(stderr_text)
193 death_summary = self._summarize_death(stderr_text)
194
195 if self._is_no_slot_error(stderr_text):
196 retry_delay = self._register_no_slot_failure(runtime)
197 runtime.next_spawn_at = max(runtime.next_spawn_at, now + retry_delay)
198 self._log(
199 f"no slot {runtime.team}#{managed_client.instance_id} "
200 f"retry_in={retry_delay:.1f}s reason={reason} "
201 f"active={len(runtime.active_clients)}/{runtime.target_count}"
202 )
203 return
204
205 runtime.no_slot_failures = 0
206 runtime.next_spawn_at = max(runtime.next_spawn_at, now + self.retry_interval)
207 if death_summary is not None:
208 self._log(
209 f"died {runtime.team}#{managed_client.instance_id} "
210 f"{death_summary} age={age:.1f}s "
211 f"active={len(runtime.active_clients)}/{runtime.target_count}"
212 )
213 self._log(
214 f"stopped {runtime.team}#{managed_client.instance_id} "
215 f"code={return_code} age={age:.1f}s reason={reason} "
216 f"active={len(runtime.active_clients)}/{runtime.target_count}"
217 )

219 async def _stop_all_clients(self) -> None:
220 self._stopping = True
221 active_processes: list[asyncio.subprocess.Process] = []
222 for runtime in self._team_runtimes.values():
223 for managed_client in runtime.active_clients.values():
224 active_processes.append(managed_client.process)
225
226 if not active_processes:
227 return
228
229 self._log(f"stopping {len(active_processes)} client process(es)")
230
231 for process in active_processes:
232 if process.returncode is None:
233 process.terminate()
234
235 await asyncio.sleep(0.5)
236
237 for process in active_processes:
238 if process.returncode is None:
239 process.kill()
240
241 await asyncio.gather(*(process.wait() for process in active_processes), return_exceptions=True)
242 await asyncio.gather(*self._watch_tasks, return_exceptions=True)

269 def _summarize_stderr(self, stderr_text: str) -> str:

274 def _summarize_death(self, stderr_text: str) -> str | None:
275 death_match = CLIENT_DEATH_RE.search(stderr_text)
276 if death_match is None:
277 return None
278 level = death_match.group(1)
279 food = death_match.group(2)
280 if food is None:
281 return f"level={level}"
282 return f"level={level} food={food}"

284 def _is_no_slot_error(self, stderr_text: str) -> bool:

287 def _register_no_slot_failure(self, runtime: TeamRuntime) -> float:
288 retry_delay = min(
289 self.max_no_slot_retry_interval,
290 self.no_slot_retry_interval * (2 ** runtime.no_slot_failures),
291 )
292 runtime.no_slot_failures += 1
293 return retry_delay

307def resolve_target_per_team(
308 *,
309 requested_target: int | None,
310 initial_clients_per_team: int,
311) -> int:
312 if requested_target == 0:
313 return UNBOUNDED_TARGET_PER_TEAM
314 if requested_target is not None:
