144void MainMenu::appendChar(int codepoint)
145{
146 if (_focus == Field::Host)
147 {
148 if (codepoint > 32 && codepoint < 127 && _host.size() < 253)
149 _host.push_back(static_cast<char>(codepoint));

151 else if (_focus == Field::Port)
152 {
153 if (codepoint >= '0' && codepoint <= '9' && _port.size() < 5)
154 _port.push_back(static_cast<char>(codepoint));

159{

185bool MainMenu::handleClick(double px, double py, MenuConfig &config)
186{
187 float fx = static_cast<float>(px);
188 float fy = static_cast<float>(_window.height()) - static_cast<float>(py);
189
190 if (contains(hostField(), fx, fy))
191 _focus = Field::Host;
192 else if (contains(portField(), fx, fy))
193 _focus = Field::Port;
194 else if (contains(modeTwoButton(), fx, fy))
195 _mode = RenderMode::TwoD;
196 else if (contains(modeThreeButton(), fx, fy))
197 _mode = RenderMode::ThreeD;
198 else if (contains(modeTorusButton(), fx, fy))
199 _mode = RenderMode::ThreeDTorus;
200 else if (contains(validateButton(), fx, fy))
201 return tryValidate(config);
202 else if (contains(quitButton(), fx, fy))
203 {
204 config = MenuConfig{MenuResult::Quit, _host, 0, _mode};
205 return true;
206 }
207 else
208 _focus = Field::None;
209 return false;
210}

212void MainMenu::drawRect(const Rect &rect, const Vec3 &color)
213{
214 Shader *shader = _assets->shader("basic");
215 Mesh *mesh = _assets->mesh("quad");
216 Mat4 model = glm::translate(Mat4(1.0f), Vec3(rect.x + rect.width / 2.0f, rect.y + rect.height / 2.0f, 0.0f));
217
218 if (shader == nullptr || mesh == nullptr)
219 return;
220 model = glm::scale(model, Vec3(rect.width, rect.height, 1.0f));
221 shader->use();
222 shader->setUniform("uView", Mat4(1.0f));
223 shader->setUniform("uProjection", glm::ortho(0.0f, static_cast<float>(_window.width()), 0.0f, static_cast<float>(_window.height())));
224 shader->setUniform("uModel", model);
225 shader->setUniform("uColor", color);
226 mesh->draw();
227}

273MenuConfig MainMenu::run()
274{
275 MenuConfig config{MenuResult::Quit, _host, 0, _mode};
276
277 while (_window.isOpen())
278 {
279 std::vector<Event> events = _window.pollEvents();
280
281 for (const Event &event : events)
282 {
283 if (event.type == EventType::Close)
284 return MenuConfig{MenuResult::Quit, _host, 0, _mode};
285 if (event.type == EventType::Resize)
286 _text->setScreenSize(event.width, event.height);
287 else if (event.type == EventType::Char)
288 appendChar(event.key);
289 else if (event.type == EventType::KeyPress && event.key == Backspace)
290 {
291 if (_focus == Field::Host && !_host.empty())
292 _host.pop_back();
293 else if (_focus == Field::Port && !_port.empty())
294 _port.pop_back();
295 }
296 else if (event.type == EventType::KeyPress && event.key == Tab)
297 _focus = (_focus == Field::Host) ? Field::Port : Field::Host;
298 else if (event.type == EventType::KeyPress && event.key == Enter)
299 {
300 if (tryValidate(config))
301 return config;
302 }
303 else if (event.type == EventType::MouseButton && event.key == 0 && event.pressed)
304 {
305 if (handleClick(event.mouseX, event.mouseY, config))
306 return config;
307 }
308 }
309 draw();
310 }
311 return MenuConfig{MenuResult::Quit, _host, 0, _mode};
312}
