52position Player::nextPosition(std::pair<int, int> mapSize) const
53{

128int Player::getDistanceTo(const position &target, std::pair<int, int> mapSize) const
129{
130 int dx = std::abs(target.x - _pos.x);
131 int dy = std::abs(target.y - _pos.y);
132
133 if (dx > (mapSize.first - dx))
134 dx = mapSize.first - dx;
135 if (dy > (mapSize.second - dy))
136 dy = mapSize.second - dy;

145Degrees Player::getDirectionTo(const position &target, std::pair<int, int> mapSize) const
146{
147 if (_pos == target)
148 return Degrees::NORTH;
149 int dx = target.x - _pos.x;
150 int dy = target.y - _pos.y;
151 if (std::abs(dx) > (mapSize.first - std::abs(dx)))
152 dx = mapSize.first - std::abs(dx);
153 if (std::abs(dy) > (mapSize.second - std::abs(dy)))
154 dy = mapSize.second - std::abs(dy);
155 return Direction::getNearestDirection(std::atan2(dy, dx) * 180 / M_PI);
156}

196void Player::sendMessageToClient()
197{
198 if (_messagesToSend.size() == 0)
199 return;
200 std::clock_t currentTime = std::clock();
201 for (auto &msg : _messagesToSend)
202 {
203 std::vector<std::pair<std::pair<std::clock_t, int>, int>> &times = msg.second;
204 for (auto it = times.begin(); it != times.end(); it++)
205 if (currentTime - it->first.first >= it->first.second * CLOCKS_PER_SEC / 1000)
206 {
207 if (it->second < 0)
208 throw ServerException("Invalid file descriptor for player");
209 ssize_t bytesSent = write(it->second, msg.first.c_str(), msg.first.size());
210 if (bytesSent < 0)
211 throw ServerException("Failed to send message to client");
212 it = times.erase(it);
213 }
214 else
215 break;
216 }
217}
