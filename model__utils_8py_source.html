48def normalize_visible_tiles(

115def select_visible_target(
116 visible_tiles: Sequence[Mapping[str, object]],
117 target_resources: set[str],
118) -> Mapping[str, object] | None:
119 best: tuple[int, int, int] | None = None
120 best_tile: Mapping[str, object] | None = None
121
122 for tile in visible_tiles:
123 if int(tile["distance"]) == 0:
124 continue
125 resources = set(tile["items"])
126 if not (resources & target_resources):
127 continue
128
129 dx = int(tile["x"])
130 distance = int(tile["distance"])
131 lateral_penalty = abs(dx)
132 candidate = (distance, lateral_penalty, int(tile["index"]))
133 if best is None or candidate < best:
134 best = candidate
135 best_tile = tile
136
137 return best_tile
