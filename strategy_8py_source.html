1"""Flat finite-state strategy for a single Zappy AI player.
2
3Each turn rebuilds a fresh Observation (Look + Inventory), picks exactly one
4state, and runs that state's body. There is no shared mutable booleans
5threaded across functions: every state body only reads the Observation and
6a handful of plain attributes on the strategy itself (rally tracking,
7cooldowns).
8
9Coordination model: a player who has decent food and at least one stone
10useful for its next elevation broadcasts a "ready" ping that lists which of
11those stones it carries. Every player tracks the ready pings it has heard
12for its own level and builds a predicted pooled inventory from them. Only
13once enough ready players are known AND their combined stones (plus our own)
14would satisfy the elevation does anyone actually move to converge - this
15avoids dragging people across the whole map towards a group that turns out
16to be incomplete once everyone arrives.
17"""

106 def step(self) -> bool:
107 if self.client.pending_level is not None:
108 # Pushed by the server because we were a participant in someone
109 # else's successful incantation, not the one who triggered it.
110 self.level = self.client.pending_level
111 self.client.pending_level = None
112 self._clear_rally()
113
114 obs = Observation(self.client, self.level)
115 if obs.dead:
116 return False
117
118 self._age_ready_seen()
119 self._process_messages(obs)
120 self.broadcast_cooldown = max(0, self.broadcast_cooldown - 1)
121 self.fork_cooldown = max(0, self.fork_cooldown - 1)
122
123 self.state = self._select_state(obs)
124 self._run_state(obs)
125 return True

129 def _select_state(self, obs: Observation) -> State:
130 if obs.food <= SURVIVAL_FOOD_THRESHOLD:
131 # Food always wins: a stale rally is worthless to a dead player.
132 self._clear_rally()
133 return State.SURVIVING
134
135 required = self._required_players()
136
137 local_group_threshold = self._local_group_threshold(required)
138
139 if not self.rallying and obs.players_here >= local_group_threshold:
140 # Already standing next to someone (often survivors of the
141 # incantation that just leveled us up): try to consolidate right
142 # here first, no broadcast or travel needed for that.
143 self.rallying = True
144 self.rally_turns = 0
145 self.target_token = self.token
146 elif not self.rallying and self._is_ready(obs) and self._enough_to_converge(obs, required):
147 self.rallying = True
148 self.rally_turns = 0
149 self.target_token = self._pick_target_token()
150
151 if self.rallying:
152 direction = self._target_direction()
153 if direction is None:
154 # Lost track of our target (its ping expired): try another
155 # known one rather than giving up immediately.
156 self.target_token = self._pick_target_token()
157 direction = self._target_direction()
158 # "Arrived" means physically grouped with at least someone, not
159 # the full quorum: stop and let the cluster grow from here
160 # instead of requiring every member to land on the same tile in
161 # the same instant, which never happens with direction-only nav.
162 fresh_target_bearing = self._target_bearing_is_fresh()
163 partial_group = obs.players_here >= local_group_threshold and not fresh_target_bearing
164 arrived = direction is None or direction == 0 or partial_group
165 if arrived and self._ready_for_incantation(obs) and obs.players_here >= required:
166 return State.INCANTING
167
168 self.rally_turns += 1
169 if self.rally_turns > self._rally_give_up_limit(required):
170 # No distance information exists in a broadcast: the pooled
171 # group turned out to be unreachable, or short a resource
172 # nobody actually had. Stop chasing it and start fresh.
173 self._clear_rally()
174 elif arrived:
175 return State.WAITING
176 else:
177 return State.JOINING
178
179 if self._ready_for_incantation(obs) and obs.players_here >= required:
180 return State.INCANTING
181 return State.GATHERING

183 def _clear_rally(self) -> None:
184 self.rallying = False
185 self.rally_turns = 0
186 self.target_token = None
187 self.ready_seen.clear()

189 def _run_state(self, obs: Observation) -> None:
190 if self.state == State.SURVIVING:
191 self._do_surviving(obs)
192 elif self.state == State.JOINING:
193 self._do_joining(obs)
194 elif self.state == State.WAITING:
195 self._do_waiting(obs)
196 elif self.state == State.INCANTING:
197 self._do_incanting(obs)
198 elif self.state == State.GATHERING:
199 self._do_gathering(obs)
200 else:
201 self._do_exploring()

203 def _ready_for_incantation(self, obs: Observation) -> bool:
204 requirement = LEVEL_REQUIREMENTS.get(self.level)
205 if requirement is None:
206 return False
207 if obs.ready_for_elevation:
208 return True
209 # Stones already pooled/left on the ground count too.
210 return ground_matches_requirement(obs.current_counts, requirement["stones"])

212 def _required_players(self) -> int:
213 requirement = LEVEL_REQUIREMENTS.get(self.level)
214 return int(requirement["players"]) if requirement else 1
215
216 def _local_group_threshold(self, required_players: int) -> int:
217 if required_players >= 6:
218 # At high levels, stopping on every pair fragments the swarm into
219 # tiny camps. A 3+ cluster is worth preserving and growing.
220 return max(2, required_players // 2)
221 return 2
222
223 def _rally_give_up_limit(self, required_players: int) -> int:
224 if required_players >= 6:
225 return RALLY_GIVE_UP_TURNS * 2
226 return RALLY_GIVE_UP_TURNS
227
228 def _is_ready(self, obs: Observation) -> bool:
229 requirement = LEVEL_REQUIREMENTS.get(self.level)
230 if requirement is None or obs.food < GATHER_FOOD_THRESHOLD:
231 return False
232 return any(obs.inv.get(stone, 0) > 0 for stone in requirement["stones"])
233
234 def _own_contribution(self, obs: Observation) -> dict[str, int]:
235 requirement = LEVEL_REQUIREMENTS.get(self.level)
236 if requirement is None:
237 return {}
238 return {
239 stone: min(obs.inv.get(stone, 0), amount)
240 for stone, amount in requirement["stones"].items()
241 if obs.inv.get(stone, 0) > 0
242 }
243
244 def _enough_to_converge(self, obs: Observation, required_players: int) -> bool:
245 if len(self.ready_seen) + 1 < required_players:
246 return False
247 requirement = LEVEL_REQUIREMENTS.get(self.level)
248 if requirement is None:
249 return False
250 pooled = Counter(self._own_contribution(obs))
251 for info in self.ready_seen.values():
252 pooled.update(info["resources"])
253 return all(pooled.get(stone, 0) >= amount for stone, amount in requirement["stones"].items())
254
255 def _pick_target_token(self) -> str:
256 # A deterministic rule (not "whoever I heard most recently") so that
257 # everyone in the same ready cohort independently agrees on the same
258 # single meeting point instead of each locking onto a different peer.
259 return min([*self.ready_seen, self.token])
260
261 def _target_direction(self) -> int | None:
262 if self.target_token is None:
263 return None
264 if self.target_token == self.token:
265 return 0
266 info = self.ready_seen.get(self.target_token)
267 return int(info["direction"]) if info else None

271 def _process_messages(self, obs: Observation) -> None:
272 for direction, text in obs.messages:
273 parsed = parse_rally_message(text)
274 if parsed is None:
275 continue
276 msg_level, msg_team, msg_token, msg_resources = parsed
277 if msg_team != self.team or msg_level != self.level or msg_token == self.token:
278 continue
279 self.ready_seen[msg_token] = {
280 "direction": direction,
281 "age": 0,
282 "resources": msg_resources,
283 }

285 def _age_ready_seen(self) -> None:
286 expired = [
287 token
288 for token, info in self.ready_seen.items()
289 if int(info["age"]) + 1 > RALLY_CALL_EXPIRY_TURNS
290 ]
291 for token in expired:
292 del self.ready_seen[token]
293 for info in self.ready_seen.values():
294 info["age"] = int(info["age"]) + 1

298 def _do_surviving(self, obs: Observation) -> None:
299 if obs.on_my_tile("food"):
300 self.client.take("food")
301 return
302 target = select_visible_target(obs.tiles, {"food"})
303 if target is not None:
304 self._move_to(target)
305 return
306 # No food in sight: scan while moving instead of committing to one
307 # straight line, vision is too narrow to risk walking away from food.
308 self._do_exploring()

310 def _do_gathering(self, obs: Observation) -> None:
311 # Forking and advertising readiness are cheap side actions, not
312 # exclusive with the main gathering move below in the same turn.
313 self._maybe_fork(obs)
314 if self._is_ready(obs):
315 self._send_ready_ping(obs)
316
317 # Below the gather threshold, food takes priority over stones even
318 # when a stone is closer: running out mid-stone-hunt is worse.
319 if obs.food < GATHER_FOOD_THRESHOLD:
320 if obs.on_my_tile("food"):
321 self.client.take("food")
322 return
323 target = select_visible_target(obs.tiles, {"food"})
324 if target is not None:
325 self._move_to(target)
326 return
327
328 resource = pick_current_tile_resource(obs.current_counts, obs.needed_stones | {"food"})
329 if resource is not None:
330 self.client.take(resource)
331 return
332 target = select_visible_target(obs.tiles, obs.needed_stones | {"food"})
333 if target is not None:
334 self._move_to(target)
335 return
336 self._do_exploring()

338 def _send_ready_ping(self, obs: Observation) -> None:
339 if self.broadcast_cooldown > 0:
340 return
341 message = build_rally_message(
342 level=self.level,
343 team=self.team,
344 token=self.token,
345 resources=self._own_contribution(obs),
346 )
347 self.client.broadcast(message)
348 self.broadcast_cooldown = BROADCAST_COOLDOWN_TURNS

350 def _do_waiting(self, obs: Observation) -> None:
351 self._take_food_if_here(obs)
352 # Keep advertising our position while waiting (even once everything
353 # has already been deposited): stragglers still need a fresh bearing
354 # to navigate by, and our entry would otherwise expire for them.
355 self._send_ready_ping(obs)
356 requirement = LEVEL_REQUIREMENTS.get(self.level)
357 if requirement is None:
358 return
359 required_players = int(requirement["players"])
360 if required_players >= 6 and obs.players_here < required_players:
361 # Keep stones in inventories while a high-level rally is still
362 # forming. Dropping them into 2-4 player camps makes broadcasts
363 # turn into "none" and strands resources across the map.
364 return
365 for stone, amount in requirement["stones"].items():
366 missing_on_ground = amount - obs.current_counts.get(stone, 0)
367 to_place = max(0, min(missing_on_ground, obs.inv.get(stone, 0)))
368 for _ in range(to_place):
369 self.client.set_obj(stone)
370
371 def _do_incanting(self, obs: Observation) -> None:
372 requirement = LEVEL_REQUIREMENTS[self.level]
373 for stone, amount in requirement["stones"].items():
374 missing = amount - obs.current_counts.get(stone, 0)
375 for _ in range(max(0, missing)):
376 self.client.set_obj(stone)
377
378 new_level = self.client.incantation()
379 if new_level is not None:
380 self.level = new_level
381 self._clear_rally()
382 # On failure the stones stay on the ground; next turn's fresh
383 # Observation re-evaluates readiness from scratch.
384
385 def _do_joining(self, obs: Observation) -> None:
386 # Grab food and any useful stone in passing, but never detour off
387 # the path to the rally: only what's already on the current tile.
388 self._take_food_if_here(obs)
389 resource = pick_current_tile_resource(obs.current_counts, obs.needed_stones)
390 if resource is not None:
391 self.client.take(resource)
392 direction = self._target_direction()
393 if direction is None:
394 return
395
396 if self._target_bearing_is_fresh():
397 # A direction is only meaningful relative to our facing at the
398 # moment it was heard. Re-applying the same turn+move every turn
399 # without a new ping would spin in place instead of approaching.
400 for command in build_plan_from_sound_direction(direction):
401 _MOVE_COMMANDS[command](self.client)
402 else:
403 self.client.forward()
404
405 def _target_bearing_is_fresh(self) -> bool:
406 if self.target_token == self.token:
407 return False
408 info = self.ready_seen.get(self.target_token)
409 return info is not None and int(info["age"]) == 0

411 def _take_food_if_here(self, obs: Observation) -> None:
412 if obs.on_my_tile("food"):
413 self.client.take("food")

415 def _do_exploring(self) -> None:
416 self._explore_tick += 1
417 if self._explore_tick % EXPLORE_TURN_INTERVAL == 0:
418 command = self._explore_turn_command
419 self._explore_turn_command = "Right" if command == "Left" else "Left"
420 _MOVE_COMMANDS[command](self.client)
421 return
422 self.client.forward()

424 def _maybe_fork(self, obs: Observation) -> bool:
425 if self.fork_cooldown > 0 or obs.food < FORK_FOOD_THRESHOLD:
426 return False
427 # An unhatched egg is already waiting: laying another one would just
428 # pile up eggs nobody connects to (the launcher caps client count).
429 if self.client.connect_nbr() > 0:
430 self.fork_cooldown = FORK_COOLDOWN_TURNS
431 return False
432 self.client.fork()
433 self.fork_cooldown = FORK_COOLDOWN_TURNS
434 return True
