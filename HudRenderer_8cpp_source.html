77{
78 float fx = static_cast<float>(px);
79 float fy = static_cast<float>(_height) - static_cast<float>(py);
80 int base = state.timeUnit() < MinTimeUnit ? MinTimeUnit : state.timeUnit();

83 return CommandBuilder::setTimeUnit(std::max(MinTimeUnit, base / 2));
