12Model ModelLoader::load(const std::string &path) const
13{
14 cgltf_options options{};
15 cgltf_data *raw = nullptr;
16
17 if (cgltf_parse_file(&options, path.c_str(), &raw) != cgltf_result_success)
18 throw ModelLoaderException("ModelLoader error: cannot parse " + path);
19
20 std::unique_ptr<cgltf_data, decltype(&cgltf_free)> document(raw, &cgltf_free);
21
22 if (cgltf_load_buffers(&options, document.get(), path.c_str()) != cgltf_result_success)
23 throw ModelLoaderException("ModelLoader error: cannot load buffers of " + path);
24 if (cgltf_validate(document.get()) != cgltf_result_success)
25 throw ModelLoaderException("ModelLoader error: invalid glTF document " + path);
26
27 Model model;
28 std::vector<std::pair<std::size_t, std::size_t>> meshRanges = readMeshes(*document, model);
29
30 readNodes(*document, meshRanges, model);
31 readSkins(*document, model);
32 readAnimations(*document, model);
33 return model;
34}

36std::vector<std::pair<std::size_t, std::size_t>> ModelLoader::readMeshes(const cgltf_data &data, Model &model) const
37{
38 std::vector<std::pair<std::size_t, std::size_t>> ranges;
39
40 for (std::size_t mesh = 0; mesh < data.meshes_count; ++mesh)
41 {
42 std::size_t first = model.primitives.size();
43
44 for (std::size_t prim = 0; prim < data.meshes[mesh].primitives_count; ++prim)
45 model.primitives.push_back(readPrimitive(data.meshes[mesh].primitives[prim], model));
46 ranges.push_back({first, model.primitives.size() - first});
47 }
48 return ranges;
49}

51ModelPrimitive ModelLoader::readPrimitive(const cgltf_primitive &primitive, Model &model) const
52{
53 ModelPrimitive result;
54
55 result.positions = readAttribute(primitive, cgltf_attribute_type_position, Vec3Size);
56 result.normals = readAttribute(primitive, cgltf_attribute_type_normal, Vec3Size);
57 result.texcoords = readAttribute(primitive, cgltf_attribute_type_texcoord, Vec2Size);
58 result.joints = readAttribute(primitive, cgltf_attribute_type_joints, Vec4Size);
59 result.weights = readAttribute(primitive, cgltf_attribute_type_weights, Vec4Size);
60 result.texture = resolveTexture(primitive, model);
61 if (primitive.indices != nullptr)
62 {
63 result.indices.reserve(primitive.indices->count);
64 for (std::size_t i = 0; i < primitive.indices->count; ++i)
65 result.indices.push_back(static_cast<unsigned int>(cgltf_accessor_read_index(primitive.indices, i)));
66 }
67 else
68 {
69 std::size_t vertices = result.positions.size() / Vec3Size;
70
71 for (std::size_t i = 0; i < vertices; ++i)
72 result.indices.push_back(static_cast<unsigned int>(i));
73 }
74 return result;
75}

77std::vector<float> ModelLoader::readAttribute(const cgltf_primitive &primitive, cgltf_attribute_type type, std::size_t components) const
78{
79 std::vector<float> values;
80
81 for (std::size_t a = 0; a < primitive.attributes_count; ++a)
82 {
83 const cgltf_attribute &attribute = primitive.attributes[a];
84
85 if (attribute.type != type || attribute.index != 0 || attribute.data == nullptr)
86 continue;
87 values.resize(attribute.data->count * components);
88 for (std::size_t i = 0; i < attribute.data->count; ++i)
89 cgltf_accessor_read_float(attribute.data, i, &values[i * components], components);
90 return values;
91 }
92 return values;
93}

95int ModelLoader::resolveTexture(const cgltf_primitive &primitive, Model &model) const
96{
97 if (primitive.material == nullptr)
98 return -1;
99
100 const cgltf_material &material = *primitive.material;
101 const cgltf_texture *texture = nullptr;
102
103 if (material.has_pbr_metallic_roughness)
104 texture = material.pbr_metallic_roughness.base_color_texture.texture;
105 if (texture == nullptr && material.has_pbr_specular_glossiness)
106 texture = material.pbr_specular_glossiness.diffuse_texture.texture;
107 if (texture == nullptr || texture->image == nullptr || texture->image->uri == nullptr)
108 return -1;
109
110 std::string uri = texture->image->uri;
111
112 for (std::size_t i = 0; i < model.textures.size(); ++i)
113 if (model.textures[i] == uri)
114 return static_cast<int>(i);
115 model.textures.push_back(uri);
116 return static_cast<int>(model.textures.size() - 1);
117}

119void ModelLoader::readNodes(const cgltf_data &data, const std::vector<std::pair<std::size_t, std::size_t>> &meshRanges, Model &model) const
120{
121 model.nodes.resize(data.nodes_count);
122 for (std::size_t n = 0; n < data.nodes_count; ++n)
123 {
124 const cgltf_node &source = data.nodes[n];
125 ModelNode &node = model.nodes[n];
126 float matrix[16] = {};
127
128 if (source.name != nullptr)
129 node.name = source.name;
130 cgltf_node_transform_local(&source, matrix);
131 node.transform = glm::make_mat4(matrix);
132 if (source.has_translation)
133 node.translation = glm::vec3(source.translation[0], source.translation[1], source.translation[2]);
134 if (source.has_rotation)
135 node.rotation = glm::quat(source.rotation[3], source.rotation[0], source.rotation[1], source.rotation[2]);
136 if (source.has_scale)
137 node.scale = glm::vec3(source.scale[0], source.scale[1], source.scale[2]);
138 if (source.mesh != nullptr)
139 {
140 std::size_t mesh = static_cast<std::size_t>(source.mesh - data.meshes);
141 std::pair<std::size_t, std::size_t> range = meshRanges[mesh];
142
143 for (std::size_t p = 0; p < range.second; ++p)
144 node.primitives.push_back(range.first + p);
145 }
146 if (source.skin != nullptr)
147 node.skin = static_cast<int>(source.skin - data.skins);
148 for (std::size_t c = 0; c < source.children_count; ++c)
149 node.children.push_back(static_cast<std::size_t>(source.children[c] - data.nodes));
150 if (source.parent == nullptr)
151 model.roots.push_back(n);
152 }
153}

155void ModelLoader::readSkins(const cgltf_data &data, Model &model) const
156{
157 model.skins.resize(data.skins_count);
158 for (std::size_t s = 0; s < data.skins_count; ++s)
159 {
160 const cgltf_skin &source = data.skins[s];
161 ModelSkin &skin = model.skins[s];
162
163 skin.joints.reserve(source.joints_count);
164 skin.inverseBind.assign(source.joints_count, Mat4(1.0f));
165 for (std::size_t j = 0; j < source.joints_count; ++j)
166 {
167 skin.joints.push_back(static_cast<std::size_t>(source.joints[j] - data.nodes));
168 if (source.inverse_bind_matrices != nullptr)
169 {
170 float matrix[Mat4Size] = {};
171
172 cgltf_accessor_read_float(source.inverse_bind_matrices, j, matrix, Mat4Size);
173 skin.inverseBind[j] = glm::make_mat4(matrix);
174 }
175 }
176 }

179void ModelLoader::readAnimations(const cgltf_data &data, Model &model) const
180{
181 model.animations.reserve(data.animations_count);
182 for (std::size_t a = 0; a < data.animations_count; ++a)
183 {
184 const cgltf_animation &source = data.animations[a];
185 ModelAnimation animation;
186
187 if (source.name != nullptr)
188 animation.name = source.name;
189 for (std::size_t c = 0; c < source.channels_count; ++c)
190 {
191 const cgltf_animation_channel &sourceChannel = source.channels[c];
192 const cgltf_animation_sampler *sampler = sourceChannel.sampler;
193 ModelAnimationChannel channel;
194
195 if (sourceChannel.target_node == nullptr || sampler == nullptr)
196 continue;
197 if (sourceChannel.target_path == cgltf_animation_path_type_translation)
198 channel.path = AnimationPath::Translation;
199 else if (sourceChannel.target_path == cgltf_animation_path_type_rotation)
200 channel.path = AnimationPath::Rotation;
201 else if (sourceChannel.target_path == cgltf_animation_path_type_scale)
202 channel.path = AnimationPath::Scale;
203 else
204 continue;
205 if (sampler->interpolation == cgltf_interpolation_type_step)
206 channel.interp = AnimationInterp::Step;
207 else if (sampler->interpolation == cgltf_interpolation_type_cubic_spline)
208 channel.interp = AnimationInterp::Cubic;
209 else
210 channel.interp = AnimationInterp::Linear;
211 channel.node = static_cast<std::size_t>(sourceChannel.target_node - data.nodes);
212 channel.components = cgltf_num_components(sampler->output->type);
213 channel.times.resize(sampler->input->count);
214 for (std::size_t k = 0; k < sampler->input->count; ++k)
215 {
216 cgltf_accessor_read_float(sampler->input, k, &channel.times[k], 1);
217 animation.duration = std::max(animation.duration, channel.times[k]);
218 }
219 channel.values.resize(sampler->output->count * channel.components);
220 for (std::size_t k = 0; k < sampler->output->count; ++k)
221 cgltf_accessor_read_float(sampler->output, k, &channel.values[k * channel.components], channel.components);
222 animation.channels.push_back(std::move(channel));
223 }
224 model.animations.push_back(std::move(animation));
225 }
