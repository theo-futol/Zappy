200void RenderSystem::render(const GameState &state)
201{

398bool RenderSystem::groundTile(double px, double py, const GameState &state, int &tileX, int &tileY) const
399{
400 Mat4 view = (_freeFly && !_following) ? _free.view() : _orbit->view();
401 Mat4 projection = (_freeFly && !_following) ? _free.projection() : _orbit->projection();
402 Mat4 viewProj = projection * view;
403 float width = static_cast<float>(_width);
404 float height = static_cast<float>(_height);
405 float bestDepth = std::numeric_limits<float>::max();
406 bool found = false;
407
408 for (int gridY = 0; gridY < state.map().height(); ++gridY)
409 for (int gridX = 0; gridX < state.map().width(); ++gridX)
410 {
411 WorldPoint point = _projection->toWorld(gridX, gridY);
412 Vec4 clip = viewProj * Vec4(point.position, 1.0f);
413
414 if (clip.w <= 0.0f)
415 continue;
416
417 float screenX = (clip.x / clip.w * 0.5f + 0.5f) * width;
418 float screenY = (1.0f - (clip.y / clip.w * 0.5f + 0.5f)) * height;
419 float dx = screenX - static_cast<float>(px);
420 float dy = screenY - static_cast<float>(py);
421
422 if (dx * dx + dy * dy > PickRadiusPx * PickRadiusPx)
423 continue;
424 if (clip.w < bestDepth)
425 {
426 bestDepth = clip.w;
427 tileX = gridX;
428 tileY = gridY;
429 found = true;
430 }
431 }
432 return found;
433}
